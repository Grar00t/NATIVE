cmake_minimum_required(VERSION 3.20)
project(niyah_search LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(niyah_search STATIC niyah_crawler.c niyah_index.c niyah_url.c
    html_extract.cpp ../native/niyah_telemetry.c)
target_include_directories(niyah_search PUBLIC . ../native)
if(MSVC)
    target_compile_definitions(niyah_search PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(niyah_search PRIVATE /W4 /WX)
else()
    target_compile_definitions(niyah_search PRIVATE _POSIX_C_SOURCE=200809L)
    target_compile_options(niyah_search PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    target_link_libraries(niyah_search PUBLIC m)
endif()
enable_testing()
foreach(component IN ITEMS url index)
    add_executable(test_niyah_${component} niyah_${component}_test.c)
    target_link_libraries(test_niyah_${component} PRIVATE niyah_search)
    add_test(NAME niyah_${component} COMMAND test_niyah_${component})
endforeach()

add_executable(test_html_extract html_extract_test.cpp)
target_link_libraries(test_html_extract PRIVATE niyah_search)
add_test(NAME niyah_html_extract COMMAND test_html_extract)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(test_index_retry niyah_index_failure_test.c)
    target_link_libraries(test_index_retry PRIVATE niyah_search)
    target_link_options(test_index_retry PRIVATE "-Wl,--wrap=realloc")
    add_test(NAME niyah_index_retry COMMAND test_index_retry)
endif()
